# mycli Node Runtime M0 Contract Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Establish the npm workspace and one language-neutral, versioned gateway contract consumed by both the current Python runtime and the TypeScript TUI.

**Architecture:** Canonical JSON Schema Draft 2020-12 files live in `packages/contracts`. A deterministic generator produces TypeScript declarations and byte-identical packaged Python copies. Python keeps its current public constants and `gateway_event_payload_schemas()` API, but loads them from the generated contract instead of defining a second schema. The TUI validates JSON-RPC envelopes and gateway notifications through the contracts package.

**Tech Stack:** Node.js 22.19+, npm workspaces, TypeScript 5.9, ESM, Ajv 2020, `json-schema-to-typescript`, ESLint, Node test runner, Python 3.13, pytest, `jsonschema` for transition-only parity tests.

---

## Scope

M0 migrates the contract surface that the repository already maintains explicitly:

- protocol version;
- supported RPC method names;
- supported gateway event names;
- gateway error codes;
- approval decision choices;
- terminal turn states;
- all current gateway event payload schemas;
- generic JSON-RPC request, response, and notification envelopes.

Per-method request parameter schemas are added when each capability migrates. M0 must not infer
new request requirements from handler implementation and accidentally freeze undocumented behavior.

```python
from __future__ import annotations

import json
from pathlib import Path
import shutil

from mycli.domain.runtime.gateway_contract import (
    APPROVAL_DECISION_CHOICES,
    GATEWAY_ERROR_CODES,
    SUPPORTED_GATEWAY_EVENT_STREAMS,
    SUPPORTED_GATEWAY_RPC_METHODS,
    TERMINAL_TURN_STATES,
    gateway_event_payload_schemas,
)


REPO_ROOT = Path(__file__).resolve().parents[3]
SCHEMA_ROOT = REPO_ROOT / "packages" / "contracts" / "schemas"
PYTHON_ROOT = REPO_ROOT / "src" / "mycli" / "schemas" / "generated"


def write_json(path: Path, value: object) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        json.dumps(value, ensure_ascii=True, indent=2, sort_keys=True) + "\n",
        encoding="utf-8",
    )


def main() -> None:
    payload_schemas = gateway_event_payload_schemas()
    event_names = sorted(SUPPORTED_GATEWAY_EVENT_STREAMS)
    catalog = {
        "$schema": "./catalog.schema.json",
        "protocolVersion": 1,
        "rpcMethods": sorted(SUPPORTED_GATEWAY_RPC_METHODS),
        "eventStreams": event_names,
        "errorCodes": list(GATEWAY_ERROR_CODES),
        "approvalDecisionChoices": list(APPROVAL_DECISION_CHOICES),
        "terminalTurnStates": list(TERMINAL_TURN_STATES),
    }
    events = {
        "$schema": "https://json-schema.org/draft/2020-12/schema",
        "$id": "https://mycli.local/contracts/gateway-events.schema.json",
        "title": "GatewayEventNotification",
        "oneOf": [
            {
                "type": "object",
                "additionalProperties": False,
                "required": ["jsonrpc", "method", "params"],
                "properties": {
                    "jsonrpc": {"const": "2.0"},
                    "method": {"const": name},
                    "params": {"$ref": f"#/$defs/{name}"},
                },
            }
            for name in event_names
        ],
        "$defs": payload_schemas,
    }
    write_json(SCHEMA_ROOT / "catalog.json", catalog)
    write_json(SCHEMA_ROOT / "gateway-events.schema.json", events)
    PYTHON_ROOT.mkdir(parents=True, exist_ok=True)
    for name in ("catalog.json", "gateway-events.schema.json"):
        shutil.copyfile(SCHEMA_ROOT / name, PYTHON_ROOT / name)


if __name__ == "__main__":
    main()
```

Create `packages/contracts/scripts/generate.mjs`:

```javascript
import {mkdir, readFile, writeFile} from "node:fs/promises";
import {dirname, relative, resolve} from "node:path";
import {fileURLToPath} from "node:url";
import {compileFromFile} from "json-schema-to-typescript";

const scriptRoot = dirname(fileURLToPath(import.meta.url));
const contractRoot = resolve(scriptRoot, "..");
const repoRoot = resolve(contractRoot, "../..");
const schemaRoot = resolve(contractRoot, "schemas");
const generatedRoot = resolve(contractRoot, "src/generated");
const pythonRoot = resolve(repoRoot, "src/mycli/schemas/generated");
const check = process.argv.includes("--check");
const banner = "// Generated by packages/contracts/scripts/generate.mjs. Do not edit.\n";

async function writeOrCheck(path, content, check) {
  const normalized = content.replaceAll("\r\n", "\n").replace(/\s+$/, "") + "\n";
  if (check) {
    const current = await readFile(path, "utf8").catch(() => "");
    if (current !== normalized) {
      throw new Error(`Generated contract drift: ${relative(repoRoot, path)}`);
    }
    return;
  }
  await mkdir(dirname(path), {recursive: true});
  await writeFile(path, normalized, "utf8");
}

const targets = [
  ["catalog.schema.json", "catalog.ts"],
  ["json-rpc.schema.json", "json-rpc-message.ts"],
  ["gateway-events.schema.json", "gateway-event-notification.ts"],
];

for (const [schemaName, outputName] of targets) {
  const generated = await compileFromFile(resolve(schemaRoot, schemaName), {
    bannerComment: "",
    cwd: schemaRoot,
    unknownAny: false,
  });
  await writeOrCheck(resolve(generatedRoot, outputName), banner + generated, check);
}

for (const name of ["catalog.json", "gateway-events.schema.json"]) {
  const content = await readFile(resolve(schemaRoot, name), "utf8");
  await writeOrCheck(resolve(pythonRoot, name), content, check);
}
```

```typescript
import {readFileSync} from "node:fs";
import type {ErrorObject, ValidateFunction} from "ajv";
import Ajv2020 from "ajv/dist/2020.js";
import type {GatewayContractCatalog} from "./generated/catalog.ts";
import type {GatewayEventNotification} from "./generated/gateway-event-notification.ts";
import type {JsonRpcMessage} from "./generated/json-rpc-message.ts";

const ajv = new Ajv2020({allErrors: true, strict: true});
ajv.addKeyword({keyword: "name", schemaType: "string", valid: true});

function compile(name: string): ValidateFunction {
  const url = new URL(`../schemas/${name}`, import.meta.url);
  return ajv.compile(JSON.parse(readFileSync(url, "utf8")) as object);
}

const validateCatalog = compile("catalog.schema.json");
const validateGatewayEvent = compile("gateway-events.schema.json");
const validateJsonRpcMessage = compile("json-rpc.schema.json");

export class ContractValidationError extends Error {
  readonly errors: readonly ErrorObject[];

  constructor(message: string, errors: readonly ErrorObject[] = []) {
    super(message);
    this.name = "ContractValidationError";
    this.errors = errors;
  }
}

function parse<T>(value: unknown, validator: ValidateFunction, label: string): T {
  if (!validator(value)) {
    throw new ContractValidationError(`Invalid ${label}.`, validator.errors ?? []);
  }
  return value as T;
}

export function parseGatewayEvent(value: unknown): GatewayEventNotification {
  return parse(value, validateGatewayEvent, "gateway event");
}

export function parseGatewayContractCatalog(value: unknown): GatewayContractCatalog {
  return parse(value, validateCatalog, "gateway contract catalog");
}

export function parseJsonRpcMessage(value: unknown): JsonRpcMessage {
  return parse(value, validateJsonRpcMessage, "JSON-RPC message");
}
```
